// Update the assignDelivery method in admin.controller.ts
// Add mongoose import at the top if not present
import mongoose from "mongoose";

// Then update the method:
assignDelivery = async (req: Request, res: Response, next: NextFunction) => {
    try {
        const { id } = req.params;
        const { deliveryPartnerId } = req.body;
        
        const order = await Order.findById(id);
        if (!order) {
            throw new AppError("Order not found", 404, ErrorCodes.ORDER_NOT_FOUND);
        }
        
        // Convert strings to ObjectId
        order.deliveryPartnerId = new mongoose.Types.ObjectId(deliveryPartnerId);
        order.status = "ASSIGNED";
        order.statusHistory.push({
            fromStatus: "READY_FOR_PICKUP",
            toStatus: "ASSIGNED",
            actorId: new mongoose.Types.ObjectId(req.userId),  // ✅ Fixed: Convert to ObjectId
            actorRole: "ADMIN",
            reason: `Assigned to delivery partner ${deliveryPartnerId}`,
            timestamp: new Date()
        });
        await order.save();
        
        res.json(SuccessResponse.success("Delivery assigned", order));
    } catch (error) {
        next(error);
    }
};
